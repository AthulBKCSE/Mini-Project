int getNumberInput(int max_options) {
    char input[10];
    int choice = 0;
    
    echo(); // Show user input
    getstr(input);
    noecho();
    
    choice = atoi(input); // Convert string to number
    
    // Validate input
    while (choice < 1 || choice > max_options) {
        printw("Invalid choice! Enter a number (1-%d): ", max_options);
        refresh();
        echo();
        getstr(input);
        noecho();
        choice = atoi(input);
    }
    
    return choice;
}

void initDirectories() {
    mkdir(USER_DIR, 0700);
    mkdir(LESSON_DIR, 0700);
}

void loadUsers() {
    FILE *file = fopen(USER_DIR "/users.dat", "rb");
    if (file != NULL) {
        fread(&userCount, sizeof(int), 1, file);
        fread(users, sizeof(User), userCount, file);
        fclose(file);
    } else {
        userCount = 0;
    }
}

void saveUsers() {
    FILE *file = fopen(USER_DIR "/users.dat", "wb");
    if (file != NULL) {
        fwrite(&userCount, sizeof(int), 1, file);
        fwrite(users, sizeof(User), userCount, file);
        fclose(file);
    }
}

int login() {
    clear();
    printw("========================================\n");
    printw("      CMD TypeSwift - Typing Tutor      \n");
    printw("========================================\n\n");
    
    printw("1. Login\n");
    printw("2. Create New User\n");
    printw("3. Exit\n\n");
    printw("Select an option: ");
    refresh();
    
    int choice = getch() - '0';
    
    if (choice == 3) {
        return 0;
    } else if (choice == 2) {
        createUser();
        return login();
    } else if (choice == 1) {
        char username[50];
        char password[50];
        
        echo();
        clear();
        printw("Username: ");
        getstr(username);
        printw("Password: ");
        noecho(); // Hide password input
        getstr(password);
        echo();
        
        for (int i = 0; i < userCount; i++) {
            if (strcmp(users[i].username, username) == 0 && 
                strcmp(users[i].password, password) == 0) {
                currentUser = users[i];
                return 1;
            }
        }
        
        printw("\nInvalid username or password!\n");
        printw("Press any key to continue...");
        refresh();
        getch();
        return login();
    } else {
        return login();
    }
}

void userMenu() {
    int choice;
    do {
        displayMenu();
        
        // Calculate total options (lessons + special options)
        int total_options = lessonCount + 4;
        
        // Use our new input function
        choice = getNumberInput(total_options);
        
        // Handle numbered lessons
        if (choice >= 1 && choice <= lessonCount) {
            runLesson(choice - 1);
        } 
        // Handle special options
        else if (choice == lessonCount + 1) {
            runCustomLesson();
        } else if (choice == lessonCount + 2) {
            addCustomLesson();
        } else if (choice == lessonCount + 3) {
            displayHelp();
        }
    } while (choice != lessonCount + 4);
}

void runLesson(int lessonIndex) {
    if (lessonIndex < 0 || lessonIndex >= lessonCount) {
        printw("Invalid lesson selection!\n");
        return;
    }
    
    clear();
    printw("Lesson: %s\n\n", lessons[lessonIndex].title);
    printw("Type the following text:\n\n");
    printw("%s\n\n", lessons[lessonIndex].content);
    printw("Press any key to start typing...");
    refresh();
    getch();
    
    clear();
    printw("Lesson: %s\n\n", lessons[lessonIndex].title);
    printw("Type the following text (press ESC to cancel):\n\n");
    
    char typedText[MAX_TEXT_LENGTH] = "";
    int cursorPos = 0;
    int correctChars = 0;
    int incorrectChars = 0;
    time_t startTime, endTime;
    
    time(&startTime);
    
    while (1) {
        clear();
        printw("Lesson: %s\n\n", lessons[lessonIndex].title);
        printw("Type the following text (press ESC to cancel):\n\n");
        
        printColoredText(lessons[lessonIndex].content, typedText);
        
        printw("\n\nProgress: %d/%d characters", cursorPos, (int)strlen(lessons[lessonIndex].content));
        refresh();
        
        int ch = getch();
        
        if (ch == 27) { // ESC key to exit
            clear();
            printw("Lesson canceled!\n");
            printw("Press any key to return to menu...");
            refresh();
            getch();
            return;
        }
        
        if (cursorPos < strlen(lessons[lessonIndex].content)) {
            if (ch == lessons[lessonIndex].content[cursorPos]) {
                correctChars++;
            } else {
                incorrectChars++;
            }
            typedText[cursorPos] = ch;
            typedText[cursorPos + 1] = '\0';
            cursorPos++;
        }
        
        if (cursorPos >= strlen(lessons[lessonIndex].content)) {
            break;
        }
    }
    
    time(&endTime);
    double timeTaken = difftime(endTime, startTime);
    
    // Update user stats
    for (int i = 0; i < userCount; i++) {
        if (strcmp(users[i].username, currentUser.username) == 0) {
            users[i].stats_correct += correctChars;
            users[i].stats_incorrect += incorrectChars;
            users[i].stats_time += timeTaken;
            users[i].stats_sessions++;
            currentUser = users[i];
            break;
        }
    }
    saveUsers();
    
    clear();
    printw("Lesson Completed!\n\n");
    displayStats(correctChars, incorrectChars, timeTaken);
    printw("\nPress any key to return to menu...");
    refresh();
    getch();
}

    char line[MAX_LINE_LENGTH];
    
    while (1) {
        getstr(line);
        if (strlen(line) == 0 && strlen(content) > 0 && content[strlen(content)-1] == '\n') {
            break;
        }
        if (strlen(content) + strlen(line) + 1 < MAX_TEXT_LENGTH) {
            strcat(content, line);
            strcat(content, "\n");
        } else {
            printw("\nError: Lesson content too long! Maximum %d characters allowed.\n", MAX_TEXT_LENGTH);
            refresh();
            getch();
            return;
        }
    }

    // Create filename from title (safe version)
    char filename[MAX_FILENAME_LENGTH];
    snprintf(filename, sizeof(filename), "%.*s.txt", 
            (int)(MAX_FILENAME_LENGTH - 5), title); // Ensure we don't overflow
    
    // Convert spaces to underscores and lowercase
    for (int i = 0; filename[i]; i++) {
        if (filename[i] == ' ') filename[i] = '_';
        filename[i] = tolower(filename[i]);
    }
    
    // Create numbered title safely
    char numberedTitle[50];
    int max_title_len = sizeof(numberedTitle) - 10; // Reserve space for number and formatting
    if (strlen(title) > max_title_len) {
        title[max_title_len] = '\0'; // Truncate if necessary
    }
    snprintf(numberedTitle, sizeof(numberedTitle), "%d. %s", lessonCount+1, title);
    
    // Save to file
    char path[MAX_PATH_LENGTH];
    snprintf(path, sizeof(path), "%s/%s", LESSON_DIR, filename);
    
    FILE *file = fopen(path, "w");
    if (file != NULL) {
        fputs(content, file);
        fclose(file);
        
        // Update lesson array
        if (lessonCount < MAX_LESSONS) {
            strcpy(lessons[lessonCount].title, numberedTitle);
            strcpy(lessons[lessonCount].filename, filename);
            strcpy(lessons[lessonCount].content, content);
            lessonCount++;
            
            printw("\nLesson saved successfully as %s!\n", filename);
        } else {
            printw("\nError: Maximum number of lessons reached!\n");
        }
    } else {
        printw("\nError: Could not save to %s\n", path);
    }
    
    printw("\nPress any key to return to menu...");
    refresh();
    getch();
}

void displayStats(int correct, int incorrect, double timeTaken) {
    int totalChars = correct + incorrect;
    double accuracy = (totalChars > 0) ? ((double)correct / totalChars) * 100 : 0;
    double wpm = (timeTaken > 0) ? (correct / 5.0) / (timeTaken / 60.0) : 0;
    double cpm = (timeTaken > 0) ? (correct / (timeTaken / 60.0)) : 0;
    
    printw("Typing Statistics:\n");
    printw("------------------\n");
    printw("Time taken: %.2f seconds\n", timeTaken);
    printw("Total characters: %d\n", totalChars);
    printw("Correct characters: %d\n", correct);
    printw("Incorrect characters: %d\n", incorrect);
    printw("Accuracy: %.2f%%\n", accuracy);
    printw("Typing speed: %.2f CPM (characters per minute)\n", cpm);
    printw("Typing speed: %.2f WPM (words per minute)\n", wpm);
    
    printw("\nYour Overall Stats:\n");
    printw("------------------\n");
    printw("Total sessions: %d\n", currentUser.stats_sessions);
    printw("Total correct characters: %d\n", currentUser.stats_correct);
    printw("Total incorrect characters: %d\n", currentUser.stats_incorrect);
    printw("Total time spent: %.2f minutes\n", currentUser.stats_time / 60.0);
    
    if (currentUser.stats_sessions > 0) {
        double overallAccuracy = ((double)currentUser.stats_correct / 
                                (currentUser.stats_correct + currentUser.stats_incorrect)) * 100;
        double avgWPM = (currentUser.stats_time > 0) ? 
                       (currentUser.stats_correct / 5.0) / (currentUser.stats_time / 60.0) : 0;
        
        printw("Overall accuracy: %.2f%%\n", overallAccuracy);
        printw("Average speed: %.2f WPM\n", avgWPM);
    }
}

void printColoredText(const char *original, const char *typed) {
    int typedLen = strlen(typed);
    
    for (int i = 0; i < strlen(original); i++) {
        if (i < typedLen) {
            if (original[i] == typed[i]) {
                attron(COLOR_PAIR(1)); // Green for correct
            } else {
                attron(COLOR_PAIR(2)); // Red for incorrect
            }
        }
        
        printw("%c", original[i]);
        
        if (i < typedLen) {
            attroff(COLOR_PAIR(1));
            attroff(COLOR_PAIR(2));
        }
    }
}

void createUser() {
    if (userCount >= MAX_USERS) {
        printw("Maximum number of users reached!\n");
        return;
    }
    
    echo();
    clear();
    printw("Create New User\n\n");
    
    char username[50];
    char password[50];
    int isAdmin = 0;
    
    printw("Username: ");
    getstr(username);
    
    // Check if username exists
    for (int i = 0; i < userCount; i++) {
        if (strcmp(users[i].username, username) == 0) {
            printw("\nUsername already exists!\n");
            printw("Press any key to continue...");
            refresh();
            getch();
            return;
        }
    }
    
    printw("Password: ");
    noecho();
    getstr(password);
    echo();
    
    if (currentUser.isAdmin) {
        printw("\nIs this user an admin? (y/n): ");
        int ch = getch();
        if (ch == 'y' || ch == 'Y') {
            isAdmin = 1;
        }
    }
    
    // Add new user
    strcpy(users[userCount].username, username);
    strcpy(users[userCount].password, password);
    users[userCount].isAdmin = isAdmin;
    users[userCount].stats_correct = 0;
    users[userCount].stats_incorrect = 0;
    users[userCount].stats_time = 0;
    users[userCount].stats_sessions = 0;
    userCount++;
    
    saveUsers();
    
    printw("\nUser created successfully!\n");
    printw("Press any key to continue...");
    refresh();
    getch();
}

void deleteUser() {
    if (!currentUser.isAdmin) {
        printw("Only admins can delete users!\n");
        return;
    }
    
    echo();
    clear();
    printw("Delete User\n\n");
    
    if (userCount == 0) {
        printw("No users to delete!\n");
        printw("Press any key to continue...");
        refresh();
        getch();
        return;
    }
    
    printw("Select user to delete:\n");
    for (int i = 0; i < userCount; i++) {
        printw("%d. %s%s\n", i+1, users[i].username, users[i].isAdmin ? " (Admin)" : "");
    }
    printw("%d. Cancel\n\n", userCount+1);
    printw("Select option: ");
    refresh();
    
    int choice = getch() - '0';
    if (choice >= 1 && choice <= userCount) {
        // Confirm deletion
        clear();
        printw("Are you sure you want to delete user '%s'? (y/n): ", users[choice-1].username);
        int confirm = getch();
        
        if (confirm == 'y' || confirm == 'Y') {
            // Shift array to remove user
            for (int i = choice-1; i < userCount-1; i++) {
                users[i] = users[i+1];
            }
            userCount--;
            saveUsers();
            
            printw("\nUser deleted successfully!\n");
        } else {
            printw("\nDeletion canceled.\n");
        }
    } else {
        printw("\nOperation canceled.\n");
    }
    
    printw("Press any key to continue...");
    refresh();
    getch();
}
